          document.addEventListener('DOMContentLoaded', function () {
            const input = document.getElementById('publication-search');
            const count = document.getElementById('publication-count');
            const groups = Array.from(document.querySelectorAll('.publication-year-group'));
            const items = Array.from(document.querySelectorAll('[data-publication]'));
            const update = () => {
              const query = input.value.trim().toLowerCase();
              let visible = 0;
              items.forEach((item) => {
                const match = !query || item.textContent.toLowerCase().includes(query);
                item.hidden = !match;
                if (match) visible += 1;
              });
              groups.forEach((group) => {
                const hasVisible = Array.from(group.querySelectorAll('[data-publication]')).some((item) => !item.hidden);
                group.hidden = !hasVisible;
              });
              count.textContent = query ? `${visible} publication${visible === 1 ? '' : 's'} found` : `Showing all ${items.length} publications`;
            };
            input.addEventListener('input', update);
            update();
          });
